cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseNativeSaveArena LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
set(CSE_SAVE_ARENA_READER "${CSE_ROOT}/src/engine/save_arena_impl.h" CACHE FILEPATH
  "Production save reader; an archived implementation can be selected for regression evidence")
configure_file(reader_header.h.in reader_header.h @ONLY)
set(save_table_owner "${CSE_ROOT}/src/engine/host_cmd.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${save_table_owner}")
file(READ "${save_table_owner}" host_cmd_source)
string(FIND "${host_cmd_source}" "TYPEDESCRIPTION gEntityTableDescription[]" tables_start)
string(FIND "${host_cmd_source}" "void EntityPatchRead(" tables_end)
if(tables_start LESS 0 OR tables_end LESS_EQUAL tables_start)
  message(FATAL_ERROR "Save table source boundaries are missing")
endif()
math(EXPR tables_length "${tables_end} - ${tables_start}")
string(SUBSTRING "${host_cmd_source}" ${tables_start} ${tables_length} save_tables)
file(CONFIGURE OUTPUT save_tables.inc CONTENT "${save_tables}" @ONLY)

add_executable(save_arena_tests save_arena_tests.cpp)
add_executable(save_allocator_tests save_allocator_tests.cpp)
add_executable(save_lightstyle_tests save_lightstyle_tests.cpp)
target_include_directories(save_lightstyle_tests PRIVATE
  "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/cl_dll" "${CSE_ROOT}/src/pm_shared"
  "${CSE_ROOT}/src/utils/vgui/include")
if(APPLE)
  target_compile_definitions(save_lightstyle_tests PRIVATE POSIX OSX)
endif()
foreach(target save_arena_tests save_allocator_tests save_lightstyle_tests)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${CSE_ROOT}/src/public" "${CSE_ROOT}/src/common")
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  endif()
endforeach()
add_test(NAME save-arena COMMAND save_arena_tests)
add_test(NAME save-allocator COMMAND save_allocator_tests)
add_test(NAME save-lightstyles COMMAND save_lightstyle_tests)
